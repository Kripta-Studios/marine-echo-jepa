"""Separately gated saved adapted-suffix reconstruction, with independent math.

No model, context, fitting, selection or original corpus is consumed. Genuine
review provenance is supplied by the caller, never generated by this module.
Synthetic fixture receipts cannot be promoted to scientific approvals.
"""

from __future__ import annotations

import argparse
import hashlib
import json
import stat
from datetime import date, datetime
from pathlib import Path

import numpy as np

from marine_echo.evaluation import native_comparison as comparison

ROLE = "adapted_suffix"
SYNTHETIC = "SYNTHETIC_CORRECTNESS_ONLY"
REAL = "REVIEWED_PREFIX_SUFFIX_ASSESSMENT"
RECIPE = dict(comparison.RECIPE)
HORIZONS = (1, 3, 6)
QUANTILES = (0.05, 0.25, 0.5, 0.75, 0.95)
TOLERANCE = 1e-10  # Frozen numerical reconstruction tolerance; no test correction.
KNOWN_AUTHORS = {
    "01a0ef20-7397-7ba1-a98c-f59bc38ddcc1",
    "01a0f72d-927a-77a3-8b62-21d456f7ed85",
}
SOURCE_PATHS = (*comparison.SOURCE_PATHS, Path(__file__).resolve())
LOADED_SOURCES = {str(p): hashlib.sha256(p.read_bytes()).hexdigest() for p in SOURCE_PATHS}
FIXTURE_ROOT = Path(__file__).resolve().parents[3] / "evidence/ssl-suffix-reconstruction-v1"


def independent_scores(a):
    """Branch-form quantile loss, then equal dates, horizons and deployments.

    This deliberately does not call native_product.native_scores. Arrays have
    already undergone strict shape/mask/finite/query validation at admission.
    """
    forecast, truth, mask = a["predictions"], a["targets"], a["observed"]
    loss = np.zeros(truth.shape, np.float64)
    for q, quantile in enumerate(QUANTILES):
        residual = np.where(mask, truth - forecast[..., q], 0.0)
        loss += np.where(residual >= 0, quantile * residual, (quantile - 1) * residual)
    loss /= len(QUANTILES)
    daily, deployments = [], {}
    for deployment in sorted(set(a["deployment"].tolist())):
        scores = []
        for h, horizon in enumerate(HORIZONS):
            groups = {}
            for row in np.flatnonzero((a["deployment"] == deployment) & mask[:, h]):
                day = str(a["target_dates"][row, h])
                if day:
                    groups.setdefault(day, []).append(float(loss[row, h]))
            eligible = []
            for day, values in sorted(groups.items()):
                if len(values) >= RECIPE["floor"]:
                    average = float(np.mean(values))
                    eligible.append(average)
                    daily.append(
                        {
                            "deployment": deployment,
                            "horizon_intervals": horizon,
                            "target_source_date": day,
                            "rows": len(values),
                            "pinball_db": average,
                        }
                    )
            scores.append(float(np.mean(eligible)) if eligible else None)
        deployments[deployment] = {
            "horizons_pinball_db": scores,
            "primary_pinball_db": float(np.mean(scores)) if None not in scores else None,
        }
    scores = [v["primary_pinball_db"] for v in deployments.values()]
    return {
        "primary_pinball_db": float(np.mean(scores)) if scores and None not in scores else None,
        "per_deployment": deployments,
        "daily_rows": daily,
        "issued_rows": len(truth),
        "scored_per_horizon": mask.sum(axis=0).tolist(),
        "eligible_source_dates_per_horizon": [
            sum(r["horizon_intervals"] == h for r in daily) for h in HORIZONS
        ],
        "quantiles": list(QUANTILES),
        "minimum_daily_rows": RECIPE["floor"],
        "aggregation": "equal_target_source_date_then_horizon_then_deployment",
        "unit": "source_reported_conditioned_native_product_dB",
    }


def independent_bootstrap(a, methods):
    """Recreate the frozen paired draws from independently calculated daily rows.

    No calls to the original block builder/bootstrap. Empty observed blocks still
    participate; repeated sampled blocks retain multiplicity. Unsupported draws or
    zero calendar span yield null intervals, never filtered successful draws.
    """
    names, blocks, spans = list(methods), [], []
    for deployment in sorted(set(a["deployment"].tolist())):
        selected = a["deployment"] == deployment
        all_dates = set(a["target_dates"][selected].ravel().tolist()) - {""}
        observed = set(a["target_dates"][selected][a["observed"][selected]].tolist()) - {""}
        anchor = date.fromisoformat(min(all_dates)) if all_dates else None

        def block(day, anchor=anchor):
            return (date.fromisoformat(day) - anchor).days // RECIPE["block_days"]

        ids = sorted({block(d) for d in observed})
        # Each slot holds independent daily values for each model/horizon.
        slots = [[[[] for _ in HORIZONS] for _ in names] for _ in ids]
        for m, name in enumerate(names):
            for row in methods[name]["metrics"]["daily_rows"]:
                if row["deployment"] == deployment:
                    slots[ids.index(block(row["target_source_date"]))][m][
                        HORIZONS.index(row["horizon_intervals"])
                    ].append(row["pinball_db"])
        sums = np.array([[[sum(v) for v in model] for model in slot] for slot in slots])
        counts = np.array([[len(v) for v in slot[0]] for slot in slots], dtype=np.int64)
        blocks.append((sums, counts))
        spans.append(
            (date.fromisoformat(max(observed)) - date.fromisoformat(min(observed))).days
            if observed
            else None
        )
    result = {
        "seed": RECIPE["bootstrap_seed"],
        "replicates": RECIPE["bootstrap_replicates"],
        "block_hours": 48,
        "block_days": 2,
        "deployment_count": len(blocks),
        "valid_replicates": 0,
        "generated_replicates": 0,
        "not_run_replicates": 2000,
        "unsupported_replicates": 0,
        "sequence_sha256": None,
        "eligible_day_count_ranges_per_horizon": None,
        "interval_status": "NOT_ASSESSABLE",
        "date_basis": "target_source_calendar_date_proxy_not_claimed_UTC",
        "verified_absolute_time": False,
    }
    if any(v["metrics"]["primary_pinball_db"] is None for v in methods.values()):
        result["reason"] = "required_daily_horizon_deployment_floor_support_incomplete"
        return result, None
    rng = np.random.default_rng(RECIPE["bootstrap_seed"])
    digest, draws, day_counts = hashlib.sha256(), [], []
    for _ in range(RECIPE["bootstrap_replicates"]):
        sampled = rng.integers(0, len(blocks), size=len(blocks))
        digest.update(sampled.astype("<i8").tobytes())
        scores, total, supported = [], np.zeros(3, np.int64), True
        for deployment in sampled:
            sums, counts = blocks[deployment]
            sample = rng.integers(0, len(counts), size=len(counts))
            digest.update(sample.astype("<i8").tobytes())
            # Direct indexed addition differs from original bincount/matrix math.
            n = counts[sample].sum(axis=0)
            total += n
            if np.any(n == 0):
                supported = False
            else:
                scores.append((sums[sample].sum(axis=0) / n).mean(axis=1))
        day_counts.append(total)
        if supported:
            draws.append(np.mean(scores, axis=0))
    result.update(
        generated_replicates=2000,
        not_run_replicates=0,
        valid_replicates=len(draws),
        unsupported_replicates=2000 - len(draws),
        sequence_sha256=digest.hexdigest(),
        eligible_day_count_ranges_per_horizon=np.stack(
            [np.min(day_counts, axis=0), np.max(day_counts, axis=0)], axis=1
        ).tolist(),
    )
    if len(draws) != 2000 or any(v == 0 for v in spans):
        result["reason"] = "unsupported_resampled_horizon_or_zero_calendar_span"
        return result, None
    result["interval_status"] = "ASSESSABLE"
    return result, np.asarray(draws)


def assert_numerical_equal(actual, expected, path="metrics"):
    """Exact structure/discrete support and fixed absolute floating tolerance."""
    if isinstance(actual, dict):
        if not isinstance(expected, dict) or actual.keys() != expected.keys():
            raise ValueError("Reconstruction structure differs: " + path)
        for key in actual:
            assert_numerical_equal(actual[key], expected[key], path + "." + key)
    elif isinstance(actual, list):
        if not isinstance(expected, list) or len(actual) != len(expected):
            raise ValueError("Reconstruction list support differs: " + path)
        for i, (a, b) in enumerate(zip(actual, expected, strict=True)):
            assert_numerical_equal(a, b, path + f"[{i}]")
    elif isinstance(actual, float):
        if (
            not np.isfinite(actual)
            or type(expected) not in (float, int)
            or not np.isfinite(expected)
            or abs(actual - expected) > TOLERANCE
        ):
            raise ValueError("Reconstruction numeric value differs: " + path)
    elif type(actual) is not type(expected) or actual != expected:
        raise ValueError("Reconstruction support/label differs: " + path)


def regular(path):
    path = Path(path).absolute()
    for part in (path, *path.parents):
        info = part.lstat()
        if stat.S_ISLNK(info.st_mode) or getattr(info, "st_file_attributes", 0) & 0x400:
            raise ValueError("No symlink/reparse source or input paths.")
    if not path.is_file():
        raise ValueError("Regular immutable input required.")
    return path


def reconstruct(manifest_path, review_path, output_path):
    """Read admission documents first; no NPZ decoding until all gates close.

    This is author-written reconstruction software, not an independent scientific
    review. Its result retains that distinction even after approved real execution.
    """
    manifest_path, review_path = regular(manifest_path), regular(review_path)
    output = Path(output_path).absolute()
    if output.exists():
        raise FileExistsError("Fresh output required; never overwrite evidence.")
    for part in output.parents:
        if part.exists() and (
            part.is_symlink() or getattr(part.lstat(), "st_file_attributes", 0) & 0x400
        ):
            raise ValueError("No reparse output paths.")
    m, mraw = comparison._read_json(manifest_path)
    review, rraw = comparison._read_json(review_path)
    if (
        m.get("kind") != "native_prefix_suffix_reconstruction_manifest_v1"
        or m.get("role") != ROLE
        or m.get("evidence_kind") not in (SYNTHETIC, REAL)
        or m.get("recipe") != RECIPE
    ):
        raise ValueError("Separately scoped reconstruction role/evidence/frozen recipe required.")
    author = comparison._identity(m.get("implementer_session_id"), "implementer")
    coordinator = comparison._identity(m.get("coordinator_session_id"), "coordinator")
    reviewer = comparison._identity(review.get("reviewer_session_id"), "reviewer")
    if reviewer.casefold() in {s.casefold() for s in {*KNOWN_AUTHORS, author, coordinator}}:
        raise ValueError("Distinct reviewer required; author validation is not approval.")
    expected = {
        "status": "APPROVED_PREFIX_SUFFIX_RECONSTRUCTION",
        "scope": "saved_adapted_suffix_reconstruction",
        "allowed_roles": [ROLE],
        "allowed_uses": ["saved_adapted_suffix_reconstruction"],
        "evidence_kind": m["evidence_kind"],
        "implementer_session_id": author,
        "coordinator_session_id": coordinator,
    }
    if any(review.get(k) != v for k, v in expected.items()):
        raise ValueError("Exact genuine reconstruction review role/status/scope required.")
    base = manifest_path.parent
    synthetic = m["evidence_kind"] == SYNTHETIC
    if synthetic and (
        m.get("fixture_identity") != SYNTHETIC
        or not base.is_relative_to(FIXTURE_ROOT)
        or not base.name.startswith(SYNTHETIC + "-")
        or not output.is_relative_to(base)
    ):
        raise ValueError("Synthetic receipts confined to private synthetic fixtures.")
    bindings, snapshots, exact = review.get("bindings"), {}, {}
    if not isinstance(bindings, dict) or not bindings:
        raise ValueError("Exact immutable bindings required.")

    def bound(path, *, source=False):
        path = regular(path)
        key = str(path)
        if synthetic and not source and not path.is_relative_to(base):
            raise ValueError("Synthetic receipt cannot admit real archives or public data.")
        expected_hash = bindings.get(key)
        if (
            not isinstance(expected_hash, str)
            or len(expected_hash) != 64
            or any(c not in "0123456789abcdef" for c in expected_hash)
        ):
            raise ValueError("Missing exact required reconstruction binding: " + key)
        raw = path.read_bytes()
        digest = hashlib.sha256(raw).hexdigest()
        if digest != expected_hash or source and digest != LOADED_SOURCES[key]:
            raise ValueError("Stale reconstruction/source binding: " + key)
        snapshots[path], exact[key] = raw, digest
        return raw

    bound(manifest_path)
    for source in SOURCE_PATHS:
        bound(source, source=True)
    docs, paths = {}, {}
    for key in ("assessment_manifest", "assessment_execution_review", "assessment_completion"):
        paths[key] = comparison._resolve(m[key], base)
        raw = bound(paths[key])
        docs[key] = json.loads(raw.decode("utf-8"), object_pairs_hook=comparison._pairs)
    assessment, execution, completion = [
        docs[k]
        for k in ("assessment_manifest", "assessment_execution_review", "assessment_completion")
    ]
    if (
        assessment.get("kind") != "native_prefix_suffix_assessment_manifest_v1"
        or completion.get("kind") != "native_prefix_suffix_assessment_completion_v1"
        or completion.get("status") not in ("COMPLETED", "NOT_ASSESSABLE")
        or completion.get("zero_shot") is not False
        or any(
            v.get("role") != ROLE or v.get("evidence_kind") != m["evidence_kind"]
            for v in (assessment, completion)
        )
        or assessment.get("recipe") != RECIPE
        or assessment.get("reference_name") != m.get("reference_name")
    ):
        raise ValueError("Own original adapted completion/role/evidence/recipe required.")
    original_authors = {
        comparison._identity(assessment.get(k), k)
        for k in ("implementer_session_id", "coordinator_session_id")
    }
    original_reviewer = comparison._identity(execution.get("reviewer_session_id"), "reviewer")
    if (
        original_reviewer.casefold() in {s.casefold() for s in original_authors | KNOWN_AUTHORS}
        or execution.get("status") != "APPROVED_PREFIX_SUFFIX_ASSESSMENT_EXECUTION"
        or execution.get("scope") != "prefix_suffix_assessment_execution"
        or execution.get("allowed_roles") != [ROLE]
        or execution.get("evidence_kind") != m["evidence_kind"]
        or any(
            execution.get(k) != assessment.get(k)
            for k in ("implementer_session_id", "coordinator_session_id")
        )
        or execution.get("allowed_cells") != assessment.get("cells")
        or execution.get("bindings", {}).get(str(paths["assessment_manifest"]))
        != exact[str(paths["assessment_manifest"])]
    ):
        raise ValueError("Original genuine execution receipt must remain bound and distinct.")
    for key in ("assessment_manifest", "assessment_execution_review"):
        if completion.get("bindings", {}).get(str(paths[key])) != exact[str(paths[key])]:
            raise ValueError("Saved original assessment lineage differs from bound receipts.")
    for key, status, scope in (
        ("software_review", "APPROVED_PREFIX_SUFFIX_SOFTWARE", "prefix_suffix_assessment_software"),
        (
            "numeric_access_review",
            "APPROVED_PREFIX_SUFFIX_NUMERIC_ACCESS",
            "prefix_suffix_numeric_assessment",
        ),
    ):
        path = comparison._resolve(assessment[key], paths["assessment_manifest"].parent)
        original = json.loads(bound(path), object_pairs_hook=comparison._pairs)
        other_reviewer = comparison._identity(original.get("reviewer_session_id"), "reviewer")
        if (
            other_reviewer.casefold() in {s.casefold() for s in original_authors | KNOWN_AUTHORS}
            or original.get("status") != status
            or original.get("scope") != scope
            or original.get("allowed_roles") != [ROLE]
            or original.get("evidence_kind") != m["evidence_kind"]
            or any(
                original.get(k) != assessment.get(k)
                for k in ("implementer_session_id", "coordinator_session_id")
            )
            or completion.get("bindings", {}).get(str(path)) != exact[str(path)]
            or execution.get("bindings", {}).get(str(path)) != exact[str(path)]
            or original.get("bindings", {}).get(str(paths["assessment_manifest"]))
            != exact[str(paths["assessment_manifest"])]
        ):
            raise ValueError("Saved original software/access approval chain differs.")
        docs[key] = original
    freeze_path = comparison._resolve(
        assessment["selection_freeze"], paths["assessment_manifest"].parent
    )
    freeze = json.loads(bound(freeze_path), object_pairs_hook=comparison._pairs)
    access = docs["numeric_access_review"]
    if (
        freeze.get("kind") != "native_prefix_suffix_selection_freeze_v1"
        or freeze.get("cells") != assessment.get("cells")
        or freeze.get("selection") != "original_development_only"
        or freeze.get("suffix_selection") is not False
        or access.get("allowed_uses") != ["frozen_adapted_suffix_assessment"]
        or access.get("selection_freeze_sha256") != exact[str(freeze_path)]
        or completion.get("bindings", {}).get(str(freeze_path)) != exact[str(freeze_path)]
    ):
        raise ValueError("Original prospective freeze/access binding differs.")
    frozen_at, admitted_at = [
        datetime.fromisoformat(v) for v in (freeze["frozen_at"], access["issued_at"])
    ]
    if frozen_at.tzinfo is None or admitted_at.tzinfo is None or frozen_at >= admitted_at:
        raise ValueError("Original freeze must precede numerical access, with explicit timezone.")
    cells = assessment.get("cells")
    if not isinstance(cells, list) or len(cells) < 2:
        raise ValueError("At least two frozen cells required.")
    names = [c.get("name") for c in cells]
    if (
        any(
            not isinstance(n, str)
            or not n
            or any(
                ch not in "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789_-"
                for ch in n
            )
            for n in names
        )
        or len(set(names)) != len(names)
        or m["reference_name"] not in names
        or set(completion.get("results", {})) != set(names)
        or len({c.get("deployment") for c in cells}) != 1
    ):
        raise ValueError("Exact original cell/deployment/reference set required.")
    for cell in cells:
        if cell.get("kind") in ("persistence", "seasonal24"):
            if cell.get("zero_shot") is not True:
                raise ValueError("Unchanged conventional reference must retain zero-shot label.")
        elif cell.get("kind") in (
            "native_prefix_matched_transfer_inference_v4",
            "native_prefix_lightgbm_inference_v1",
        ):
            if cell.get("zero_shot") is not False or cell.get("prefix_days") not in (1, 7, 30):
                raise ValueError("Adapted kind must retain prefix budget and adaptation label.")
        else:
            raise ValueError("Only enumerated original saved adapted-suffix cell kinds.")
    if not any(c.get("zero_shot") is False for c in cells):
        raise ValueError("At least one adapted cell required in adaptation reconstruction.")
    partition = completion.get("partition", {})
    fit_rows, suffix_rows = partition.get("fit_rows"), partition.get("suffix_rows")
    fit_ids, suffix_ids = partition.get("fit_interval_ids"), partition.get("suffix_interval_ids")
    if (
        not isinstance(fit_rows, list)
        or not isinstance(suffix_rows, list)
        or not isinstance(fit_ids, list)
        or not isinstance(suffix_ids, list)
        or not suffix_rows
        or not suffix_ids
        or any(not isinstance(r, list) or len(r) != 2 for r in fit_rows + suffix_rows)
        or len(set(map(tuple, suffix_rows))) != len(suffix_rows)
        or set(map(tuple, fit_rows)) & set(map(tuple, suffix_rows))
        or set(fit_ids) & set(suffix_ids)
    ):
        raise ValueError("Original prefix/suffix row and interval partition must remain disjoint.")
    predictions = {name: paths["assessment_completion"].parent / (name + ".npz") for name in names}
    if set(completion.get("prediction_hashes", {})) != {str(p) for p in predictions.values()}:
        raise ValueError("Saved full prediction inventory differs; no intersections.")
    for name, path in predictions.items():
        bound(path)
        if completion["prediction_hashes"][str(path)] != exact[str(path)]:
            raise ValueError("Saved prediction digest differs from original completion.")
        if type(completion["results"][name].get("zero_shot")) is not bool or (
            completion["results"][name]["zero_shot"] != cells[names.index(name)].get("zero_shot")
        ):
            raise ValueError("Original per-cell adaptation/zero-shot labels differ.")
    # Admission is now complete; only exact saved forecast artifacts are decoded.
    common, metrics = None, {}
    for name, path in predictions.items():
        a, _, _ = comparison._load(snapshots[path], ROLE, m["evidence_kind"])
        if (
            a.get("cutoff") is None
            or a["cutoff"].shape != (len(a["targets"]),)
            or a["cutoff"].dtype.kind not in "iu"
            or np.any(a["cutoff"] < 0)
            or set(a["deployment"].tolist()) != {cells[0]["deployment"]}
            or set(zip(a["deployment"].tolist(), a["row_id"].tolist(), strict=True))
            != set(map(tuple, suffix_rows))
        ):
            raise ValueError("Saved cutoff/deployment identity required.")
        if not np.allclose(a["query"][..., 3:5] * 250, [0, 230], atol=1e-5, rtol=0):
            raise ValueError("Original suffix native 0-230m geometry must be retained.")
        if "query_native_bounds_m" in a and not np.allclose(
            a["query_native_bounds_m"], a["query"][..., 3:5] * 250, atol=1e-5, rtol=0
        ):
            raise ValueError("Saved geometry alias differs from actual native queries.")
        if common is not None:
            comparison._common(common, a)
            if not np.array_equal(common["cutoff"], a["cutoff"]):
                raise ValueError("Exact common cutoff identities differ.")
        else:
            common = a
        actual = independent_scores(a)
        assert_numerical_equal(actual, completion["results"][name]["metrics"], name)
        metrics[name] = {"metrics": actual, "zero_shot": completion["results"][name]["zero_shot"]}
    if "native_geometry" in completion:
        assert_numerical_equal(
            comparison._geometry(common["query"]), completion["native_geometry"], "native_geometry"
        )
    uncertainty, draws = independent_bootstrap(common, metrics)
    for key, value in uncertainty.items():
        if key in completion.get("uncertainty", {}):
            assert_numerical_equal(value, completion["uncertainty"][key], "uncertainty." + key)
        elif key not in ("verified_absolute_time", "date_basis"):
            raise ValueError("Missing original uncertainty support: " + key)
    ref = m["reference_name"]
    paired = {}
    for i, name in enumerate(names):
        score, reference = [metrics[n]["metrics"]["primary_pinball_db"] for n in (name, ref)]
        paired[name] = {
            "reference": ref,
            "method_minus_reference": None
            if score is None or reference is None
            else score - reference,
            "interval": None
            if draws is None
            else np.percentile(draws[:, i] - draws[:, names.index(ref)], [2.5, 97.5]).tolist(),
        }
    assert_numerical_equal(paired, completion.get("paired"), "paired")
    required_status = (
        "COMPLETED"
        if all(v["metrics"]["primary_pinball_db"] is not None for v in metrics.values())
        else "NOT_ASSESSABLE"
    )
    if completion["status"] != required_status:
        raise ValueError("Saved completion assessability differs from reconstructed support.")
    report = {
        "kind": "native_prefix_suffix_reconstruction_completion_v1",
        "status": "RECONSTRUCTED",
        "assessment_status": required_status,
        "role": ROLE,
        "zero_shot": False,
        "evidence_kind": m["evidence_kind"],
        "results": metrics,
        "paired": paired,
        "uncertainty": uncertainty,
        "frozen_cells": cells,
        "original_partition": partition,
        "native_geometry": comparison._geometry(common["query"]),
        "bindings": exact,
        "review_sha256": hashlib.sha256(rraw).hexdigest(),
        "manifest_sha256": hashlib.sha256(mraw).hexdigest(),
        "reviewer_session_id": reviewer,
        "independent_scientific_approval": False,
        "absolute_tolerance": TOLERANCE,
        "reconstruction_scope": "saved forecasts/support/daily scores/paired bootstrap; no weight replay",
        "scientific_claim": None,
    }
    with output.open("x", encoding="utf-8") as stream:
        json.dump(report, stream, indent=2, allow_nan=False)
        stream.write("\n")
    return report


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    for name in ("manifest", "review", "output"):
        parser.add_argument("--" + name, required=True)
    args = parser.parse_args()
    reconstruct(args.manifest, args.review, args.output)


if __name__ == "__main__":
    main()
